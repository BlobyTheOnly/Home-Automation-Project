#include "DHT.h"
#include <WiFi.h>
#include <PubSubClient.h>
 
// =====================================================
// SENSOR AND ACTUATOR PINS
// =====================================================
 
#define DHT_PIN 15
#define DHT_TYPE DHT22
#define PIR_PIN 27
#define LDR_PIN 34
#define ROOM_LED 2
#define FAN_LED 4
#define BUZZER 5
 
// =====================================================
// AUTOMATION SETTINGS
// =====================================================
 
#define DARK_THRESHOLD 2000
#define FAN_ON_TEMP 28.0
#define FAN_OFF_TEMP 26.0
#define HIGH_TEMP 35.0
#define NO_MOTION_TIMEOUT 30000UL
 
// =====================================================
// WIFI SETTINGS - WOKWI
// =====================================================
 
const char* WIFI_SSID = "Wokwi-GUEST";
const char* WIFI_PASSWORD = "";
 
// =====================================================
// MQTT SETTINGS
// =====================================================
 
const char* MQTT_SERVER = "broker.hivemq.com";
const int MQTT_PORT = 1883;
 
// MQTT topics
const char* TOPIC_TEMPERATURE = "smartroom/temperature";
const char* TOPIC_HUMIDITY    = "smartroom/humidity";
const char* TOPIC_LIGHT       = "smartroom/light";
const char* TOPIC_MOTION      = "smartroom/motion";
const char* TOPIC_FAN         = "smartroom/fan";
const char* TOPIC_LED         = "smartroom/led";
const char* TOPIC_BUZZER      = "smartroom/buzzer";
 
// =====================================================
// OBJECTS
// =====================================================
 
DHT dht(DHT_PIN, DHT_TYPE);
 
WiFiClient espClient;
PubSubClient mqttClient(espClient);
 
// =====================================================
// VARIABLES
// =====================================================
 
unsigned long lastMotionTime = 0;
unsigned long lastMqttAttempt = 0;
 
bool fanState = false;
 
// =====================================================
// WIFI CONNECTION
// =====================================================
 
void setupWiFi() {
 
  Serial.println();
  Serial.println("Connecting to Wi-Fi...");
 
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
 
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
 
  Serial.println();
  Serial.println("Wi-Fi connected!");
 
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
}
 
// =====================================================
// MQTT CONNECTION
// =====================================================
 
void maintainMQTT() {
 
  // If Wi-Fi is disconnected, don't try MQTT
  if (WiFi.status() != WL_CONNECTED) {
    return;
  }
 
  // If MQTT is already connected
  if (mqttClient.connected()) {
    mqttClient.loop();
    return;
  }
 
  // Try to reconnect every 5 seconds
  if (millis() - lastMqttAttempt >= 5000) {
 
    lastMqttAttempt = millis();
 
    Serial.println("Connecting to MQTT broker...");
 
    // Create a unique client ID
    String clientId = "SmartRoom-";
    clientId += String((uint32_t)ESP.getEfuseMac(), HEX);
 
    if (mqttClient.connect(clientId.c_str())) {
 
      Serial.println("MQTT connected!");
 
    } else {
 
      Serial.print("MQTT connection failed. State: ");
      Serial.println(mqttClient.state());
    }
  }
}
 
// =====================================================
// PUBLISH SENSOR DATA
// =====================================================
 
void publishMQTT(
  float temperature,
  float humidity,
  int lightLevel,
  bool motionDetected,
  bool fanOn,
  bool ledOn,
  bool buzzerOn
) {
 
  // Only publish if MQTT is connected
  if (!mqttClient.connected()) {
    return;
  }
 
  // Convert temperature to text
  char temperatureString[10];
  dtostrf(temperature, 1, 1, temperatureString);
 
  // Convert humidity to text
  char humidityString[10];
  dtostrf(humidity, 1, 1, humidityString);
 
  // Convert light level to text
  char lightString[10];
  sprintf(lightString, "%d", lightLevel);
 
  // Publish sensor data
  mqttClient.publish(
    TOPIC_TEMPERATURE,
    temperatureString
  );
 
  mqttClient.publish(
    TOPIC_HUMIDITY,
    humidityString
  );
 
  mqttClient.publish(
    TOPIC_LIGHT,
    lightString
  );
 
  // Publish motion status
  if (motionDetected) {
    mqttClient.publish(
      TOPIC_MOTION,
      "DETECTED"
    );
  } else {
    mqttClient.publish(
      TOPIC_MOTION,
      "NO MOTION"
    );
  }
 
  // Publish fan status
  if (fanOn) {
    mqttClient.publish(
      TOPIC_FAN,
      "ON"
    );
  } else {
    mqttClient.publish(
      TOPIC_FAN,
      "OFF"
    );
  }
 
  // Publish LED status
  if (ledOn) {
    mqttClient.publish(
      TOPIC_LED,
      "ON"
    );
  } else {
    mqttClient.publish(
      TOPIC_LED,
      "OFF"
    );
  }
 
  // Publish buzzer status
  if (buzzerOn) {
    mqttClient.publish(
      TOPIC_BUZZER,
      "ON"
    );
  } else {
    mqttClient.publish(
      TOPIC_BUZZER,
      "OFF"
    );
  }
 
  Serial.println("MQTT data published.");
}
 
// =====================================================
// SETUP
// =====================================================
 
void setup() {
 
  Serial.begin(115200);
 
  // Start DHT22
  dht.begin();
 
  // Sensor pins
  pinMode(PIR_PIN, INPUT);
 
  // Actuator pins
  pinMode(ROOM_LED, OUTPUT);
  pinMode(FAN_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
 
  // Start actuators OFF
  digitalWrite(ROOM_LED, LOW);
  digitalWrite(FAN_LED, LOW);
  digitalWrite(BUZZER, LOW);
 
  // Start motion timer
  lastMotionTime = millis();
 
  // MQTT server
  mqttClient.setServer(
    MQTT_SERVER,
    MQTT_PORT
  );
 
  // Connect Wi-Fi
  setupWiFi();
 
  Serial.println("================================");
  Serial.println(" SmartRoom Stage 3 Starting...");
  Serial.println("================================");
 
  Serial.print("MQTT Broker: ");
  Serial.println(MQTT_SERVER);
 
  Serial.print("MQTT Port: ");
  Serial.println(MQTT_PORT);
 
  Serial.println();
}
 
// =====================================================
// MAIN LOOP
// =====================================================
 
void loop() {
 
  // Maintain MQTT connection
  maintainMQTT();
 
  // ===================================================
  // READ SENSORS
  // ===================================================
 
  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();
 
  int motion = digitalRead(PIR_PIN);
 
  int lightLevel = analogRead(LDR_PIN);
 
  // ===================================================
  // CHECK DHT22
  // ===================================================
 
  if (isnan(temperature) || isnan(humidity)) {
 
    Serial.println("ERROR: Failed to read DHT22");
 
    delay(2000);
 
    return;
  }
 
  // ===================================================
  // SENSOR STATES
  // ===================================================
 
  bool motionDetected = (motion == HIGH);
 
  bool roomIsDark = (lightLevel < DARK_THRESHOLD);
 
  // ===================================================
  // UPDATE MOTION TIMER
  // ===================================================
 
  if (motionDetected) {
 
    lastMotionTime = millis();
  }
 
  unsigned long timeSinceMotion =
    millis() - lastMotionTime;
 
  // ===================================================
  // RULE 1 - SMART LIGHTING
  // ===================================================
  //
  // Motion detected AND room is dark
  // -> Room LED ON
  //
 
  if (motionDetected && roomIsDark) {
 
    digitalWrite(ROOM_LED, HIGH);
  }
 
  // ===================================================
  // RULE 3 - ENERGY SAVING
  // ===================================================
  //
  // No motion for 30 seconds
  // -> Room LED OFF
  //
 
  if (timeSinceMotion >= NO_MOTION_TIMEOUT) {
 
    digitalWrite(ROOM_LED, LOW);
  }
 
  // ===================================================
  // RULE 2 - TEMPERATURE CONTROL
  // ===================================================
  //
  // Temperature above 28°C
  // -> Fan ON
  //
  // Temperature below 26°C
  // -> Fan OFF
  //
  // This creates hysteresis and prevents rapid
  // ON/OFF switching.
  //
 
  if (temperature > FAN_ON_TEMP) {
 
    fanState = true;
  }
 
  if (temperature < FAN_OFF_TEMP) {
 
    fanState = false;
  }
 
  // ===================================================
  // RULE 4 - HIGH TEMPERATURE WARNING
  // ===================================================
  //
  // Temperature above 35°C
  // -> Fan ON
  // -> Buzzer ON
  //
 
  if (temperature > HIGH_TEMP) {
 
    fanState = true;
 
    digitalWrite(BUZZER, HIGH);
 
  } else {
 
    digitalWrite(BUZZER, LOW);
  }
 
  // ===================================================
  // APPLY FAN STATE
  // ===================================================
 
  digitalWrite(
    FAN_LED,
    fanState
  );
 
  // ===================================================
  // DETERMINE ROOM STATE
  // ===================================================
 
  if (!motionDetected) {
 
    Serial.println("Room State: EMPTY");
 
  } else if (motionDetected && roomIsDark) {
 
    Serial.println("Room State: OCCUPIED & DARK");
 
  } else {
 
    Serial.println("Room State: OCCUPIED");
  }
 
  // ===================================================
  // SERIAL MONITOR
  // ===================================================
 
  Serial.println("--------------------------------");
 
  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.println(" °C");
 
  Serial.print("Humidity: ");
  Serial.print(humidity);
  Serial.println(" %");
 
  Serial.print("Motion: ");
 
  if (motionDetected) {
 
    Serial.println("DETECTED");
 
  } else {
 
    Serial.println("NO MOTION");
  }
 
  Serial.print("Light Level: ");
  Serial.println(lightLevel);
 
  Serial.print("Room Dark: ");
 
  if (roomIsDark) {
 
    Serial.println("YES");
 
  } else {
 
    Serial.println("NO");
  }
 
  Serial.print("Fan: ");
 
  if (fanState) {
 
    Serial.println("ON");
 
  } else {
 
    Serial.println("OFF");
  }
 
  Serial.print("Room LED: ");
 
  if (digitalRead(ROOM_LED) == HIGH) {
 
    Serial.println("ON");
 
  } else {
 
    Serial.println("OFF");
  }
 
  Serial.print("Buzzer: ");
 
  if (digitalRead(BUZZER) == HIGH) {
 
    Serial.println("ON");
 
  } else {
 
    Serial.println("OFF");
  }
 
  Serial.print("Time Since Motion: ");
  Serial.print(timeSinceMotion / 1000);
  Serial.println(" seconds");
 
  // ===================================================
  // MQTT
  // ===================================================
 
  publishMQTT(
    temperature,
    humidity,
    lightLevel,
    motionDetected,
    fanState,
    digitalRead(ROOM_LED) == HIGH,
    digitalRead(BUZZER) == HIGH
  );
 
  Serial.println("--------------------------------");
 
  // ===================================================
  // LOOP DELAY
  // ===================================================
 
  delay(2000);
}
